function doPost(e){
  if(e&&e.parameter&&e.parameter.tg) return tgWebhook_(e);   // сообщения от Telegram-бота
  let out;
  try{
    const req=JSON.parse(e&&e.postData&&e.postData.contents||'{}');
    const tok=token_(); if(!tok) throw Error('Скрипт не настроен — запусти setup в Apps Script');
    if(typeof req.k!=='string'||req.k!==tok){ Utilities.sleep(800); throw Error('Неверный ключ'); }
    const fn=API[req.fn]; if(!fn) throw Error('Неизвестная команда');
    migrate_();
    out={ok:true,data:fn.apply(null,Array.isArray(req.args)?req.args:[])};
  }catch(err){ out={ok:false,error:String(err&&err.message||err)}; }
  return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON);
}

function round_(v){ return Math.round(num_(v)*100)/100; }

function ymd_(v,tz){
  if(v instanceof Date) return Utilities.formatDate(v,tz,'yyyy-MM-dd');
  const s=String(v||'').trim(); let m=s.match(/^(\d{4})-(\d{2})-(\d{2})/); if(m) return m[0];
  m=s.match(/^(\d{1,2})\.(\d{1,2})\.(\d{4})$/); if(m) return m[3]+'-'+('0'+m[2]).slice(-2)+'-'+('0'+m[1]).slice(-2);
  return '';
}

const cur_=v=>String(v).trim().toUpperCase()==='AMD'?'AMD':'USD';
function readLog_(sh,tz){
  if(!sh||sh.getLastRow()<2) return [];
  return sh.getRange(2,1,sh.getLastRow()-1,HEAD.length).getValues().map((r,i)=>{
    if(!r[1]&&r[4]==='') return null;
    const e={id:String(r[0]||('row-'+(i+2))), date:ymd_(r[1],tz), type:typeOf_(r[2])||'income',
      cur:cur_(r[3]), amount:round_(r[4]), saved:round_(r[5]),
      from:keyOf_(FROM,r[6]), category:String(r[7]||''), note:String(r[8]||''), rate:round_(r[9]),
      account:String(r[10]||''), person:String(r[11]||''), src:'site'};
    const sp=split_(r[15]); if(sp) e.split=sp;
    if(e.type==='transfer') e.to={account:String(r[12]||''), cur:cur_(r[13]), amount:round_(r[14])};
    return e;
  }).filter(Boolean);

function logRow_(sh,id){
  if(sh.getLastRow()<2) return 0;
  if(/^row-\d+$/.test(id)) return Number(id.slice(4));
  const ids=sh.getRange(2,1,sh.getLastRow()-1,1).getValues();
  for(let i=0;i<ids.length;i++) if(String(ids[i][0])===id) return i+2;
  return 0;
}
function logValues_(e,tz){
  const t=e.type==='transfer'&&e.to?e.to:null;
  return [e.id,date_(e.date,tz),TYPES[e.type],e.cur,e.amount,e.type==='income'?e.saved:'',FROM[e.from]||'',literal_(e.category),literal_(e.note),e.rate>0?e.rate:'',
    literal_(e.account||''),literal_(e.person||''),t?literal_(t.account):'',t?t.cur:'',t?t.amount:'',
    e.split?JSON.stringify(e.split.map(p=>({a:p.account,v:p.amount}))):''];
}
// «Разбивка по счетам»: [{a:'Карта',v:60},{a:'Наличные',v:40}] → [{account,amount}]; null, если её нет
function split_(v){
  if(!v) return null;
  try{ const x=JSON.parse(String(v)); if(Array.isArray(x)&&x.length>1) return x.map(p=>({account:String(p.a||''),amount:round_(p.v)})); }catch(err){}
  return null;
}

// ---------- лист «Ввод» (общий, если ведёте вместе) ----------
function input_(){ const sh=book_().getSheetByName(INPUT); if(!sh) throw Error('Лист «Ввод» не найден'); return sh; }
function readInput_(sh,tz){
  const v=sh.getRange(FIRST,1,LAST-FIRST+1,7).getValues(), all=[];
  v.forEach((r,i)=>{
    if(r[0]===''&&r[1]===''&&r[2]===''&&r[3]==='') return;
    all.push({row:FIRST+i,date:ymd_(r[0],tz),who:String(r[1]).trim(),income:round_(r[2]),saved:round_(r[3]),note:String(r[5]||''),id:String(r[6]||'')});
  });
  return all;
}
// Насколько запись меняет отложенные деньги (в валюте записи).
// from: 'savings' — деньги берутся из отложенных (или кладутся в них для входящих), иначе — свободные.
function savedDelta_(e){
  const a=e.amount, sv=e.from==='savings';
  if(e.type==='income') return e.saved||0;
  if(e.type==='save') return sv?-a:a;                    // sv — достал из отложений обратно
  if(e.type==='out'||e.type==='lend'||e.type==='repay') return sv?-a:0;
  if(e.type==='lendBack'||e.type==='borrow') return sv?a:0;
  return 0;                                              // перевод между счетами не меняет отложенное
}
// Строка во «Ввод» в долларах: доход и всё, что меняет отложенное. Драмы — по курсу записи.
// Комментарий во «Ввод» всегда пустой: откуда деньги и куда ушли — только в личном журнале.
function mirror_(e){
  const amd=e.cur==='AMD';
  if(amd&&!(e.rate>0)) return null;
  const usd=v=>amd?Math.round(v/e.rate*100)/100:v;
  const c=e.type==='income'?usd(e.amount):0, d=usd(savedDelta_(e));
  if(e.type!=='income'&&!d) return null;
  return [c,d,''];
}
function inputRowById_(sh,id){
  const ids=sh.getRange(FIRST,7,LAST-FIRST+1,1).getValues();
  for(let i=0;i<ids.length;i++) if(String(ids[i][0])===id) return FIRST+i;
  return 0;
}
function nextInputRow_(sh){
  const v=sh.getRange(FIRST,1,LAST-FIRST+1,4).getValues(); let last=-1;
  v.forEach((r,i)=>{ if(r.some(x=>x!=='')) last=i; });
  const row=FIRST+last+1; if(row>LAST) throw Error('«Ввод» заполнен до '+LAST+' строки — Dashboard дальше не считает.');
  return row;
}
function writeInput_(sh,row,e,m,tz){
  sh.getRange(row,1,1,4).setValues([[date_(e.date,tz),ME,m[0],m[1]]]);
  sh.getRange(row,6,1,2).setValues([[literal_(m[2]),e.id]]);
  sh.getRange(row,7).setFontColor('#bbbbbb').setFontSize(6);
}
function clearInput_(sh,row){ sh.getRange(row,1,1,4).clearContent(); sh.getRange(row,6,1,2).clearContent(); }
function syncInput_(before,after,tz){
  const m=after?mirror_(after):null, id=(after||before).id;
  if(!m&&!(before&&mirror_(before))) return;
  const sh=input_(), row=inputRowById_(sh,id);
  if(m) writeInput_(sh,row||nextInputRow_(sh),after,m,tz);
  else if(row) clearInput_(sh,row);
}
// Один раз: журнал переезжает из общей таблицы в личный файл, из «Ввод» убираются тексты записей приложения.
function migratePrivate_(){
  if(props_().getProperty('migrated-private')==='1') return;
  locked_(()=>{
    if(props_().getProperty('migrated-private')==='1') return;
    const shared=book_(), old=shared.getSheetByName(LEGACY_LOG);
    if(old){
      const sh=log_(true), n=old.getLastRow()-1;
      if(n>0){
        const rows=old.getRange(2,1,n,HEAD.length).getValues();
        if(sh.getLastRow()<2) sh.getRange(2,1,n,HEAD.length).setValues(rows);
        SpreadsheetApp.flush();
        if(sh.getLastRow()-1!==n) throw Error('Перенос журнала не сошёлся — общий лист не тронут');
      }
      shared.deleteSheet(old);
    }
    const inp=input_(), ids=inp.getRange(FIRST,7,LAST-FIRST+1,1).getValues();
    ids.forEach((r,i)=>{ if(String(r[0])) inp.getRange(FIRST+i,6).setValue(''); });
    props_().setProperty('migrated-private','1');
  });
}

// Один раз: старые записи в драмах (до этой версии) получают курс и строку во «Ввод».
function migrate_(){
  migratePrivate_();
  if(props_().getProperty('migrated')==='amd-v2') return;
  const rate=rate_(); if(!(rate>0)) return;                 // без курса перенести нельзя — попробуем позже
  locked_(()=>{
    if(props_().getProperty('migrated')==='amd-v2') return;
    const tz=book_().getSpreadsheetTimeZone(), sh=log_(false);
    if(sh) readLog_(sh,tz).forEach(e=>{
      if(e.cur!=='AMD') return;
      if(!(e.rate>0)){ e.rate=rate; sh.getRange(logRow_(sh,e.id),10).setValue(rate); }
      if(mirror_(e)&&!inputRowById_(input_(),e.id)) syncInput_(null,e,tz);
    });
    props_().setProperty('migrated','amd-v2');
  });
}

// ---------- проверка записи ----------
function clean_(e,id,keepRate){
  if(!e||typeof e!=='object') throw Error('Пустая запись');
  if(!TYPES[e.type]) throw Error('Неизвестный тип записи');
  if(e.cur!=='USD'&&e.cur!=='AMD') throw Error('Валюта: $ или ֏');
  if(!/^\d{4}-\d{2}-\d{2}$/.test(e.date)) throw Error('Проверь дату');
  const amount=round_(e.amount), saved=e.type==='income'?round_(e.saved||0):0;
  if(!(amount>0)||amount>1e10) throw Error('Сумма должна быть больше нуля');
  if(saved<0||saved>amount) throw Error('Отложить можно от 0 до суммы дохода');
  let rate=round_(e.rate)||keepRate||rate_();            // курс на момент записи — для пересчёта $ ↔ ֏
  if(!(rate>0&&rate<100000)){ if(e.cur==='AMD') throw Error('Сначала поставь курс (֏ за 1 $) — драмы идут в таблицу в долларах'); rate=0; }
  const s=(x,n)=>String(x||'').trim().slice(0,n);
  const c={id:id||(/^[\w-]{8,64}$/.test(e.id||'')?e.id:Utilities.getUuid()), date:e.date, type:e.type, cur:e.cur,
    amount, saved, rate, from:e.type==='income'?'':(e.from==='savings'?'savings':'income'),
    category:e.type==='out'||e.type==='income'?s(e.category,40):'', note:s(e.note,300),
    account:e.type==='save'?'':s(e.account,40), person:DEBT_TYPES.indexOf(e.type)>=0?s(e.person,60):''};
  if(DEBT_TYPES.indexOf(e.type)>=0&&!c.person) throw Error('Укажи, с кем долг');
  if(Array.isArray(e.split)&&e.type!=='save'&&e.type!=='transfer'){          // часть на карту, часть наличными…
    const parts={}; e.split.forEach(p=>{ const a=s(p&&p.account,40), v=round_(p&&p.amount); if(v>0) parts[a]=round_((parts[a]||0)+v); });
    const list=Object.keys(parts).map(a=>({account:a,amount:parts[a]}));
    if(list.length>1){
      const sum=round_(list.reduce((x,p)=>x+p.amount,0));
      if(Math.abs(sum-amount)>0.009) throw Error('Разбивка по счетам ('+sum+') не равна сумме ('+amount+')');
      c.split=list; c.account=list[0].account;
    } else if(list.length===1) c.account=list[0].account;
  }
  if(e.type==='transfer'){
    const t=e.to||{}, ta=round_(t.amount), tc=t.cur==='AMD'?'AMD':t.cur==='USD'?'USD':'';
    if(!tc) throw Error('Перевод: выбери валюту, куда');
    if(!(ta>0)||ta>1e10) throw Error('Перевод: сумма, куда, должна быть больше нуля');
    c.to={account:s(t.account,40), cur:tc, amount:tc===c.cur?amount:ta};
    if(c.to.account===c.account&&tc===c.cur) throw Error('Перевод: откуда и куда совпадают');
  }
  return c;
}

function getRates_(force){
  const cache=CacheService.getScriptCache();
  if(!force){ const c=cache.get('rates'); if(c) return JSON.parse(c); }
  const res=UrlFetchApp.fetch(RATES_URL,{muteHttpExceptions:true,headers:{'User-Agent':'Mozilla/5.0','Accept-Language':'hy,ru;q=0.8'}});
  if(res.getResponseCode()!==200) throw Error('rate.am не ответил ('+res.getResponseCode()+')');
  const rates=parseRates_(res.getContentText());
  cache.put('rates',JSON.stringify(rates),1200);
  return rates;
}
function parseRates_(html){
  let flight=''; const re=/self\.__next_f\.push\(\[1,("(?:[^"\\]|\\.)*")\]\)/g; let m;
  while((m=re.exec(html))) flight+=JSON.parse(m[1]);
  const map={};
  flight.split('\n').forEach(line=>{ const x=line.match(/^([0-9a-f]+):(\{.*\})$/); if(x){ try{ map[x[1]]=JSON.parse(x[2]); }catch(err){} } });
  const ref=v=>typeof v==='string'&&v.charAt(0)==='$'?map[v.slice(1)]:v;
  const isHolder=v=>{ const h=ref(v); return !!(h&&typeof h==='object'&&'rates' in h); };
  const list=Object.values(map).map(o=>Object.entries(o||{})).find(en=>en.length>3&&en.every(([k,v])=>isHolder(v)))||[];
  const banks=[];
  list.forEach(([slug,v])=>{
    const holder=ref(v), usd=ref((ref(holder.rates)||{}).USD)||{}, cash=ref(usd.CASH)||{};
    const buy=Number(cash.buy), sell=Number(cash.sell);
    if(buy>0&&sell>0) banks.push({bank:slug.split('-').map(w=>w.charAt(0).toUpperCase()+w.slice(1)).join(' '),buy,sell,updated:Number(holder.lastUpdated)||0});
  });
  if(!banks.length) throw Error('Не удалось прочитать курсы с rate.am');
  const pick=(key,dir)=>banks.reduce((a,b)=>dir*(b[key]-a[key])>0?b:a);
  return {banks:banks.length, buyMin:pick('buy',-1), buyMax:pick('buy',1), sellMin:pick('sell',-1), sellMax:pick('sell',1),
    updated:Math.max.apply(null,banks.map(b=>b.updated))||Date.now()};
}

// Разбор первого сообщения: «2000», «50$», «+50 $ зарплата», «2к шаурма», «5000 դրամ»
const TG_RE = /^([+\-−]?)\s*([$֏])?\s*(\d[\d\s.,]*?)\s*([kкKК])?(?=\s|$|[$֏])\s*(\$|֏|(?:usd|amd|дол\S*|драм\S*|др\.?|դոլ\S*|դր\S*)(?=\s|$))?\s*([\s\S]*)$/i;
function tgParse_(t){
  const m=String(t).trim().match(TG_RE);
  if(!m) return null;
  let num=m[3].replace(/\s/g,'');
  num=/^\d{1,3}([.,]\d{3})+$/.test(num)?num.replace(/[.,]/g,''):num.replace(',','.');
  let v=parseFloat(num); if(!(v>0)) return null; if(m[4]) v*=1000; v=Math.round(v*100)/100;
  const sym=m[2]||m[5]||'', c=!sym?null:/^(\$|usd|дол|դոլ)/i.test(sym)?'USD':'AMD';
  return {amount:v, cur:c, type:m[1]==='+'?'income':m[1]?'out':null, note:m[6].trim().slice(0,300)};
}

  Object.keys(L.D).forEach(k=>{ const p=L.D[k], open=type==='lendBack'?uni_(p.USD.lent,p.AMD.lent,'USD')>0.05:type==='repay'?uni_(p.USD.owe,p.AMD.owe,'USD')>0.05:true; if(open) list.push({n:p.name,last:p.last}); });
  return list.sort((a,b)=>b.last.localeCompare(a.last)).map(x=>x.n);
}
function tgCallback_(q){
  const chat=String(q.message&&q.message.chat&&q.message.chat.id), data=String(q.data||''), mid=q.message&&q.message.message_id;
  try{ tgCall_('answerCallbackQuery',{callback_query_id:q.id}); }catch(err){}
  if(chat!==cfg_('tgChat')) return;
  if(data.indexOf('u:')===0) return tgUndo_(chat,data.slice(2),mid);
  const d=tgDraft_(chat);
  if(!d||d.msg!==mid) return tgEdit_(chat,mid,'⌛ Эта запись устарела — напиши сумму заново.');
  const k=data.split(':')[0], v=data.split(':')[1];
  if(k==='x'){ tgSave_(chat,null); return tgEdit_(chat,mid,'✖️ Отменено'); }
  if(k==='c') d.cur=v==='USD'?'USD':'AMD';
  else if(k==='t') d.type=v;
  else if(k==='p') d.person=(d.people||[])[+v]||'';
  else if(k==='P'){ d.wait='person'; tgSave_(chat,d); return tgEdit_(chat,mid,tgHead_(d)+'\n\nНапиши имя сообщением:',{inline_keyboard:[[B_('✖️ Отмена','x')]]}); }
  else if(k==='f'){ d.from=v; d.fromDone=true; }
  else if(k==='k'){ const inc=d.type==='income', list=items_(inc?'incomeCategories':'categories',inc?INCOME_CATEGORIES:CATEGORIES); d.category=v==='-'?'':((list[+v]||{}).n||''); d.catDone=true; }
  else if(k==='a'){ const a=items_('accounts',ACCOUNTS)[+v]; d.account=a?a.n:''; d.accDone=true; }
  else if(k==='s'){ d.saved=Math.round(d.amount*(+v))/100; d.savedDone=true; }
  else if(k==='S'){ d.wait='saved'; tgSave_(chat,d); return tgEdit_(chat,mid,tgHead_(d)+'\n\nСколько отложил? Напиши число:',{inline_keyboard:[[B_('✖️ Отмена','x')]]}); }
  else if(k==='n'){ d.note=''; d.noteDone=true; d.wait=''; }
  return tgNext_(chat,d);
}
function tgCommit_(chat,d){
  try{
    const e={id:Utilities.getUuid(), type:d.type, cur:d.cur, date:today_(), amount:d.amount, saved:d.saved||0,
      from:d.from||'income', category:d.category||'', note:d.note||'', account:d.type==='save'?'':(d.account||''), person:d.person||'', rate:rate_()};
    addEntry_(e); tgSave_(chat,null); setCfg_('tgLast',e.id);
    const B=ledger_(getState_().entries).B;
    tgEdit_(chat,d.msg,'✅ '+tgHead_(d)+'\n\n<i>свободно '+fmt_(uni_(B.USD.free,B.AMD.free,d.cur),d.cur)+' · отложено '+fmt_(uni_(B.USD.saved,B.AMD.saved,d.cur),d.cur)+'</i>',{inline_keyboard:[[B_('↩️ Отменить','u:'+e.id)]]});
  }catch(err){ tgSave_(chat,null); tgEdit_(chat,d.msg,'⚠️ Не записалось: '+esc_(err.message)); }
}
function tgUndo_(chat,id,msg){
  if(!id) return tgSend_(chat,'Нечего отменять.');
  try{ deleteEntry_(id); if(cfg_('tgLast')===id) setCfg_('tgLast',null); }
  catch(err){ return msg?tgEdit_(chat,msg,'Уже удалено или не найдено.'):tgSend_(chat,'Уже удалено или не найдено.'); }
  return msg?tgEdit_(chat,msg,'↩️ Отменено — запись удалена.'):tgSend_(chat,'↩️ Последняя запись удалена.');
}

// ---------- баланс и отчёты ----------
function ledger_(entries){
  const B={USD:{free:0,saved:0},AMD:{free:0,saved:0}}, A={}, D={};
  const acc=(n,c,v)=>{ A[n]=A[n]||{USD:0,AMD:0}; A[n][c]+=v; }, bk=(c,sv,v)=>{ B[c][sv?'saved':'free']+=v; };
  const ax=(e,sg)=>{ if(e.split&&e.split.length) e.split.forEach(p=>acc(p.account||'',e.cur,sg*p.amount)); else acc(e.account||'',e.cur,sg*e.amount); };
  const debt=(p,c,k,v,date)=>{ const key=String(p).trim().toLowerCase(); D[key]=D[key]||{name:String(p).trim(),USD:{lent:0,owe:0},AMD:{lent:0,owe:0},last:''}; D[key][c][k]+=v; if(date>D[key].last) D[key].last=date; };
  entries.forEach(e=>{ const a=e.amount, c=e.cur, n=e.account||'', sv=e.from==='savings';
    switch(e.type){
      case 'income': ax(e,1); B[c].saved+=e.saved||0; B[c].free+=a-(e.saved||0); break;
      case 'out': ax(e,-1); bk(c,sv,-a); break;
      case 'save': bk(c,!sv,a); bk(c,sv,-a); break;
      case 'transfer': { const t=e.to||{}; acc(n,c,-a); acc(t.account||'',t.cur||c,t.amount||a); bk(c,sv,-a); bk(t.cur||c,sv,t.amount||a); break; }
      case 'lend': ax(e,-1); bk(c,sv,-a); debt(e.person||'?',c,'lent',a,e.date); break;
      case 'lendBack': ax(e,1); bk(c,sv,a); debt(e.person||'?',c,'lent',-a,e.date); break;
      case 'borrow': ax(e,1); bk(c,sv,a); debt(e.person||'?',c,'owe',a,e.date); break;
      case 'repay': ax(e,-1); bk(c,sv,-a); debt(e.person||'?',c,'owe',-a,e.date); break; } });
  return {B:B,A:A,D:D};
}
// деньги — один котёл: пара {USD,AMD} → одна сумма в валюте cur по текущему курсу
function uni_(usd,amd,cur){ const r=rate_(); if(!(r>0)) return cur==='USD'?usd:amd; return cur==='USD'?usd+amd/r:amd+usd*r; }
function both_(usd,amd){
  const r=rate_(); if(!(r>0)) return [usd?fmt_(usd,'USD'):'',amd?fmt_(amd,'AMD'):''].filter(Boolean).join(' + ')||fmt_(0,'AMD');
  return fmt_(amd+usd*r,'AMD')+' ≈ '+fmt_(usd+amd/r,'USD');
}
function tgBalance_(chat){
  const st=getState_(), L=ledger_(st.entries), B=L.B, lines=[];
  lines.push('💼 <b>Всего: '+both_(B.USD.free+B.USD.saved,B.AMD.free+B.AMD.saved)+'</b>');
  lines.push('Свободно: '+both_(B.USD.free,B.AMD.free));
  lines.push('Отложено: '+both_(B.USD.saved,B.AMD.saved));
  lines.push('');
  st.accounts.forEach(a=>{ const v=L.A[a.n]; if(v&&Math.abs(uni_(v.USD,v.AMD,'USD'))>0.004) lines.push(a.i+' '+esc_(a.n)+': '+both_(v.USD,v.AMD)); });
  const rec={USD:0,AMD:0}, pay={USD:0,AMD:0};
  Object.keys(L.D).forEach(k=>{ const x=L.D[k], l=uni_(x.USD.lent,x.AMD.lent,'USD'), o=uni_(x.USD.owe,x.AMD.owe,'USD'); if(l>0.05) rec.USD+=l; if(o>0.05) pay.USD+=o; });
  if(rec.USD||rec.AMD) lines.push('🤝 Тебе должны: '+both_(rec.USD,rec.AMD));
  if(pay.USD||pay.AMD) lines.push('🧾 Ты должен: '+both_(pay.USD,pay.AMD));
  if(st.goal){ const r=rate_(), gc=st.goal.cur, sv=gc==='USD'?B.USD.saved+(r>0?B.AMD.saved/r:0):B.AMD.saved+B.USD.saved*r;
    lines.push('', '🎯 '+esc_(st.goal.title)+': <b>'+Math.round(Math.max(0,sv)/st.goal.amount*100)+'%</b> ('+fmt_(sv,gc)+' из '+fmt_(st.goal.amount,gc)+')'); }
  return tgSend_(chat,lines.join('\n'));
}
function tgReport_(chat,days,auto){
  const st=getState_(), tz=book_().getSpreadsheetTimeZone(), r=rate_();
  const since=Utilities.formatDate(new Date(Date.now()-(days-1)*864e5),tz,'yyyy-MM-dd');
  const list=st.entries.filter(e=>e.date>=since);
  const toU=(e,v)=>e.cur==='USD'?v:((e.rate||r)>0?v/(e.rate||r):0);
  let inc=0,out=0,sv=0; const cats={};
  list.forEach(e=>{ const a=e.amount, s=e.from==='savings'; let dv=0;
    if(e.type==='income') inc+=toU(e,a);
    if(e.type==='out'){ out+=toU(e,a); const c=e.category||'Без категории'; cats[c]=(cats[c]||0)+toU(e,a); }
    if(e.type==='income') dv=e.saved||0; else if(e.type==='save') dv=s?-a:a; else if(['out','lend','repay'].indexOf(e.type)>=0) dv=s?-a:0; else if(['lendBack','borrow'].indexOf(e.type)>=0) dv=s?a:0;
    sv+=toU(e,dv); });
  const two=v=>r>0?fmt_(v*r,'AMD')+' ≈ '+fmt_(v,'USD'):fmt_(v,'USD');
  const lines=[(auto?'📅 <b>Итоги недели</b>':'📅 <b>За '+days+' дней</b>'),'','💰 Пришло: '+two(inc),'🧾 Ушло: '+two(out),'🐖 Отложил: '+two(sv)+(inc>0?' ('+Math.round(sv/inc*100)+'% дохода)':''),'📈 Итог: '+(inc-out>=0?'+':'')+two(inc-out)];
  const top=Object.keys(cats).sort((a,b)=>cats[b]-cats[a]).slice(0,3);
  if(top.length){ lines.push('','Больше всего ушло на:'); top.forEach(c=>lines.push(esc_(tgIcon_('categories',c)+' '+c)+' — '+two(cats[c])+' ('+Math.round(cats[c]/out*100)+'%)')); }
  if(!list.length) lines.push('','Записей не было. Напиши сумму — запишу 🙂');
  return tgSend_(chat,lines.join('\n'));
}

function tgWeekly(){ const chat=cfg_('tgChat'); if(chat&&cfg_('tgWeekly')!=='0') tgReport_(chat,7,true); }

  ScriptApp.newTrigger('tgWeekly').timeBased().onWeekDay(ScriptApp.WeekDay.SUNDAY).atHour(20).inTimezone(book_().getSpreadsheetTimeZone()).create();
}
